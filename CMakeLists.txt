cmake_minimum_required(VERSION 3.18)
project(slz VERSION 0.1.0 LANGUAGES C)

option(SLZ_ASM "Use assembly CRC32 fast paths (GCC/Clang toolchains)" ON)
option(SLZ_SANITIZE "Build with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
set(SLZ_TOP OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(SLZ_TOP ON) # tests by default only when slz is the top-level project
endif()
option(SLZ_TESTS "Build tests" ${SLZ_TOP})
option(SLZ_FAULT_INJECTION "Test builds: SLZ_FAULT_AT=N fails the N-th I/O operation" OFF)
option(SLZ_INSTALL "Generate install rules" ON)
option(SLZ_ZSTD "zstd codec for the .slz format (system zstd, else fetched v1.5.7)" ON)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(SLZ_SRC src/inflate.c src/lzma.c src/lz4.c src/native.c src/sevenz.c src/crc32.c src/formats.c src/os.c src/extract.c)

# Assembly needs a GNU-style assembler (not cl.exe / ml64); MSVC falls back to portable C.
set(SLZ_ASM_SRC "")
if(SLZ_ASM AND NOT MSVC AND NOT CMAKE_SYSTEM_NAME STREQUAL "WASI" AND NOT EMSCRIPTEN)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    set(SLZ_ASM_SRC src/crc32_x86_64.S)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    set(SLZ_ASM_SRC src/crc32_aarch64.S)
  endif()
  if(SLZ_ASM_SRC)
    enable_language(ASM)
  endif()
endif()

add_library(slz ${SLZ_SRC} ${SLZ_ASM_SRC})
add_library(slz::slz ALIAS slz)
target_include_directories(slz PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(slz PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR} PUBLIC_HEADER include/slz.h)
if(SLZ_ASM_SRC)
  target_compile_definitions(slz PRIVATE SLZ_ASM)
endif()
if(MSVC)
  target_compile_options(slz PRIVATE /W4)
else()
  target_compile_options(slz PRIVATE -Wall -Wextra -Wshadow -Wconversion -Wno-sign-conversion)
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
  find_package(Threads REQUIRED)
  target_link_libraries(slz PUBLIC Threads::Threads)
endif()
# zstd: an installed package if there is one, otherwise the official 1.5.7 release (SHA-256 pinned)
set(SLZ_ZSTD_FETCHED OFF)
if(SLZ_ZSTD)
  find_package(zstd CONFIG QUIET)
  if(TARGET zstd::libzstd_static)
    set(SLZ_ZSTD_LIB zstd::libzstd_static)
  elseif(TARGET zstd::libzstd_shared)
    set(SLZ_ZSTD_LIB zstd::libzstd_shared)
  else()
    include(FetchContent)
    FetchContent_Declare(zstd
      URL https://github.com/facebook/zstd/releases/download/v1.5.7/zstd-1.5.7.tar.gz
      URL_HASH SHA256=eb33e51f49a15e023950cd7825ca74a4a2b43db8354825ac24fc1b7ee09e6fa3
      SOURCE_SUBDIR build/cmake
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
    set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
    set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(zstd)
    set(SLZ_ZSTD_LIB libzstd_static)
    set(SLZ_ZSTD_FETCHED ON)
    target_include_directories(slz PRIVATE ${zstd_SOURCE_DIR}/lib)
  endif()
  target_compile_definitions(slz PRIVATE SLZ_ZSTD)
  target_link_libraries(slz PRIVATE ${SLZ_ZSTD_LIB})
endif()
if(SLZ_FAULT_INJECTION)
  target_compile_definitions(slz PRIVATE SLZ_FAULT_INJECTION)
endif()
if(SLZ_SANITIZE)
  target_compile_options(slz PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer -g)
  target_link_options(slz PUBLIC -fsanitize=address,undefined)
endif()

add_executable(slz-cli tools/slz.c)
set_target_properties(slz-cli PROPERTIES OUTPUT_NAME slz)
target_link_libraries(slz-cli slz)

if(SLZ_TESTS)
  add_executable(slz-test tests/test_api.c)
  target_include_directories(slz-test PRIVATE src)
  target_link_libraries(slz-test slz)
  if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "WASI") # archive fuzzer (POSIX), built on demand by CI
    add_executable(fuzz_archive EXCLUDE_FROM_ALL tests/fuzz_archive.c)
    target_link_libraries(fuzz_archive slz)
  endif()
  enable_testing()
  add_test(NAME unit COMMAND slz-test unit)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    set(SLZ_TEST_ARGS)
    if(SLZ_FAULT_INJECTION)
      set(SLZ_TEST_ARGS --fault-cli $<TARGET_FILE:slz-cli>) # the same CLI is fault-enabled
    endif()
    # tests/run_tests.py split into parallel tests: archives-data builds the corpus and the reference
    # archives once, every archives-* test runs one part (or one slice of it) against them
    set(SLZ_TEST_RUN ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.py
        $<TARGET_FILE:slz-cli> $<TARGET_FILE:slz-test>)
    set(SLZ_TEST_DATA ${CMAKE_CURRENT_BINARY_DIR}/archives-data)
    add_test(NAME archives-data COMMAND ${SLZ_TEST_RUN} --prepare ${SLZ_TEST_DATA})
    add_test(NAME archives-clean COMMAND ${SLZ_TEST_RUN} --clean ${SLZ_TEST_DATA})
    set_tests_properties(archives-data PROPERTIES FIXTURES_SETUP slz-archives)
    set_tests_properties(archives-clean PROPERTIES FIXTURES_CLEANUP slz-archives)
    # COST: ctest starts the slowest parts first
    function(slz_archives_test part shards)
      cmake_parse_arguments(T "" "ONLY" "" ${ARGN})
      set(cost_destructive 90)
      set(cost_fault 50)
      set(cost_fuzz 40)
      set(cost_roundtrip 20)
      set(cost_read 20)
      set(cost_kill 10)
      set(cost_hostile 5)
      set(name archives-${part})
      set(args --part ${part})
      if(T_ONLY)
        string(APPEND name -${T_ONLY})
        list(APPEND args --only ${T_ONLY})
      endif()
      foreach(k RANGE 1 ${shards})
        set(n ${name})
        if(shards GREATER 1)
          string(APPEND n -${k}of${shards})
        endif()
        add_test(NAME ${n} COMMAND ${SLZ_TEST_RUN} ${SLZ_TEST_ARGS} --data ${SLZ_TEST_DATA} ${args} --shard ${k}/${shards})
        set_tests_properties(${n} PROPERTIES FIXTURES_REQUIRED slz-archives COST ${cost_${part}})
      endforeach()
    endfunction()
    # heavy parts last and in consecutive runs, so `ctest -I k,,N` (CI's Windows job shards) splits
    # them evenly
    slz_archives_test(roundtrip 2)
    slz_archives_test(hostile 1)
    slz_archives_test(read 1)
    slz_archives_test(fuzz 2)
    foreach(arc def9.zip pax.tar nonsolid.7z)
      slz_archives_test(kill 1 ONLY ${arc})
    endforeach()
    slz_archives_test(destructive 6)
    if(SLZ_FAULT_INJECTION)
      foreach(arc def1.zip nonsolid.7z)
        slz_archives_test(fault 3 ONLY ${arc})
      endforeach()
    endif()
  endif()
endif()

# find_package(slz) / target_link_libraries(app slz::slz) after `cmake --install`
if(SLZ_INSTALL)
  include(GNUInstallDirs)
  include(CMakePackageConfigHelpers)
  install(TARGETS slz EXPORT slzTargets
          ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
          LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
          RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
          PUBLIC_HEADER DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(TARGETS slz-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  if(SLZ_ZSTD_FETCHED) # the static slz needs the fetched zstd archive next to it
    install(TARGETS libzstd_static EXPORT slzTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  endif()
  install(EXPORT slzTargets NAMESPACE slz:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/slz)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/slzConfig.cmake
       "include(CMakeFindDependencyMacro)\nif(NOT WIN32)\n  find_dependency(Threads)\nendif()\n"
       "if(${SLZ_ZSTD} AND NOT ${SLZ_ZSTD_FETCHED})\n  find_dependency(zstd CONFIG)\nendif()\n"
       "include(\"\${CMAKE_CURRENT_LIST_DIR}/slzTargets.cmake\")\n")
  write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/slzConfigVersion.cmake COMPATIBILITY SameMajorVersion)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/slzConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/slzConfigVersion.cmake
          DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/slz)
endif()
