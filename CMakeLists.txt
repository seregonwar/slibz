cmake_minimum_required(VERSION 3.13)
project(slz C)

option(SLZ_ASM "Use assembly CRC32 fast paths (GCC/Clang toolchains)" ON)
option(SLZ_SANITIZE "Build with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
option(SLZ_TESTS "Build tests" ON)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(SLZ_SRC src/inflate.c src/lzma.c src/sevenz.c src/crc32.c src/formats.c src/os.c src/extract.c)

# Assembly needs a GNU-style assembler (not cl.exe / ml64); MSVC falls back to portable C.
set(SLZ_ASM_SRC "")
if(SLZ_ASM AND NOT MSVC AND NOT CMAKE_SYSTEM_NAME STREQUAL "WASI" AND NOT EMSCRIPTEN)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    set(SLZ_ASM_SRC src/crc32_x86_64.S)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    set(SLZ_ASM_SRC src/crc32_aarch64.S)
  endif()
  if(SLZ_ASM_SRC)
    enable_language(ASM)
  endif()
endif()

add_library(slz ${SLZ_SRC} ${SLZ_ASM_SRC})
target_include_directories(slz PUBLIC include)
if(SLZ_ASM_SRC)
  target_compile_definitions(slz PRIVATE SLZ_ASM)
endif()
if(MSVC)
  target_compile_options(slz PRIVATE /W4)
else()
  target_compile_options(slz PRIVATE -Wall -Wextra -Wshadow -Wconversion -Wno-sign-conversion)
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "WASI")
  find_package(Threads REQUIRED)
  target_link_libraries(slz PUBLIC Threads::Threads)
endif()
if(SLZ_SANITIZE)
  target_compile_options(slz PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer -g)
  target_link_options(slz PUBLIC -fsanitize=address,undefined)
endif()

add_executable(slz-cli tools/slz.c)
set_target_properties(slz-cli PROPERTIES OUTPUT_NAME slz)
target_link_libraries(slz-cli slz)

if(SLZ_TESTS)
  add_executable(slz-test tests/test_api.c)
  target_include_directories(slz-test PRIVATE src)
  target_link_libraries(slz-test slz)
  enable_testing()
  add_test(NAME unit COMMAND slz-test unit)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME archives COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.py $<TARGET_FILE:slz-cli> $<TARGET_FILE:slz-test>)
  endif()
endif()
